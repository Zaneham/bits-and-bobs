                  </PieChart>
                ) : (
                  <BarChart data={exposureData}>
                    <CartesianGrid strokeDasharray="3 3" />
                    <XAxis dataKey="name" />
                    <YAxis />
                    <Tooltip formatter={(value) => [`${value}`, 'Amount']} />
                    <Legend />
                    <Bar dataKey="value" fill="#2563eb" name="Exposure" />
                  </BarChart>
                )}
              </ResponsiveContainer>
            </section>

            {/* FX Rates Section */}
            <section className="fx-rates">
              <h2>FX Rates (vs NZD)</h2>
              <table>
                <thead>
                  <tr>
                    <th>Currency Pair</th>
                    <th>Locked Rate</th>
                    <th>Market Rate</th>
                    <th>Difference</th>
                  </tr>
                </thead>
                <tbody>
                  {fxRates.map(rate => (
                    <tr key={rate.pair}>
                      <td>{rate.pair}</td>
                      <td>{rate.locked}</td>
                      <td>{rate.market}</td>
                      <td className={rate.locked > rate.market ? 'positive' : 'negative'}>
                        {(rate.locked - rate.market).toFixed(3)}
                      </td>
                    </tr>
                  ))}
                </tbody>
              </table>
            </section>

            {/* Scenario Simulator Section */}
            <section className="scenario-section">
              <h2>Scenario Simulator</h2>
              <p className="scenario-explainer">
                Select invoices or switch to currency view to simulate your exposure. 
                You can also run canned "what-if" scenarios to see how shifts in FX rates 
                or payment delays would affect totals.
              </p>

              <div className="scenario-buttons">
                {scenarios.map(scenario => (
                  <button
                    key={scenario.id}
                    onClick={() => runScenario(scenario.id)}
                  >
                    {scenario.label}
                  </button>
                ))}
                <button
                  onClick={() => {
                    setInvoices(initialInvoices);
                    addAuditEntry("Scenarios reset to baseline");
                  }}
                >
                  Reset Scenarios
                </button>
              </div>

              <ScenarioPanel invoices={invoices} />
            </section>

            {/* Audit Trail Section */}
            <section className="audit-section">
              <h2>Audit Trail</h2>
              <p className="audit-explainer">
                Every action is logged here for transparency, invoice changes, scenario runs, 
                FX refreshes, and exports. This ensures a clear, auditable history of decisions.
              </p>
              <div className="audit-trail">
                {auditLog.length === 0 ? (
                  <p className="no-entries">No actions recorded yet</p>
                ) : (
                  auditLog.map((entry, idx) => (
                    <div key={idx} className="audit-entry">
                      {entry}
                    </div>
                  ))
                )}
              </div>
            </section>
          </div>
        </div>
      </div>
    </>
  );
}
